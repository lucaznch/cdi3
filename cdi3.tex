\section{Introdução}
O foco principal de Cálculo Diferencial e Integral III são as equações diferenciais.
Estes resumos apresentam métodos de resolução para os seus diferentes tipos.
Faz sentido começar por algumas definições.

\subsection{Equação Diferencial}
Uma equação diferencial é uma equação que envolve uma função desconhecida e as suas derivadas.

As incógnitas destas equações são funções, por isso
as soluções das equações diferenciais vão ser sempre funções.

\subsection{Equação Diferencial Ordinária (EDO)}
A função desconhecida depende de uma única variável independente.

$$
\frac{d^2 x}{dt^2} + \omega^2 x = 0 \quad \text{que corresponde a} \quad x'' + \omega^2 x = 0
$$

\begin{itemize}
    \item A função desconhecida é $x(t)$.
    \item A função $x(t)$ é uma variável dependente que depende apenas da variável independente $t$.
\end{itemize}

\subsection{Equação Diferencial Parcial (EDP)}
A função desconhecida depende de várias variáveis independentes.

$$
\frac{\partial u}{\partial t} = \alpha \frac{\partial^2 u}{\partial x^2}
$$

\begin{itemize}
    \item A função desconhecida é $u(x,t)$.
    \item A função $u(x,t)$ é uma variável dependente que depende das variáveis independentes $x$ e $t$.
\end{itemize}

\subsection{Ordem de uma Equação Diferencial}
A ordem de uma equação diferencial é a maior ordem de derivada presente na equação.

$$
y''' + 3y'' - 4y' + 7y = 0
$$

\begin{itemize}
    \item A equação diferencial é de ordem 3.
\end{itemize}

\subsection{Equação Diferencial Ordinária Linear e Não Linear}
Uma equação diferencial ordinária de ordem $n$ é dita linear se pode ser escrita na forma:

$$
a_0(t) y^{(n)} + a_1(t) y^{(n-1)} + \ldots + a_{n-1}(t) y' + a_n(t) y = g(t)
$$

onde $a_n(t), a_{n-1}(t), \ldots, a_1(t), a_0(t)$ e $g(t)$ são funções conhecidas.\par
\bigskip
Se a equação não pode ser escrita nesta forma, é dita não linear.

\subsubsection{Equação Diferencial Ordinária Linear Homogénea e Não Homogénea}
Uma equação diferencial ordinária linear de ordem $n$ pode ser classificada como:

\begin{itemize}
    \item Homogénea se $g(t) = 0$.
    \item Não Homogénea se $g(t) \neq 0$.
\end{itemize}

\section{Equações Diferenciais Ordinárias de Primeira Ordem}

\subsection{Equação Diferencial Ordinária de 1ª Ordem Linear Homogénea}
\label{subsec: edo-ordem1-linear-homogenea}

\begin{align*}
& \boxed{y' + a(t) y = 0} \\
& y' = -a(t) y \\
& \frac{y'}{y} = -a(t) \\
& \frac{d}{dt} \Big[ \ln|y| \Big] = -a(t) \\
& \ln|y| = \int -a(t) dt \\
& \ln|y| = - \int a(t) dt + c, \quad c \in \mathbb{R} \\
& |y| = e^{-\int a(t) dt + c}, \quad c \in \mathbb{R} \\
& |y| = e^c e^{-\int a(t) dt}, \quad c \in \mathbb{R} \\
& |y| = k e^{-\int a(t) dt}, \quad k \in \mathbb{R}^+ \text{ pois } k = e^c > 0 \\
& y = \pm k e^{-\int a(t) dt}, \quad k \in \mathbb{R}^+ \\
& y = k e^{-\int a(t) dt}, \quad k \in \mathbb{R} \setminus \{0\} \\
& \text{Como } y(t) = 0 \text{ é também solução, conclui-se que:} \\
& \boxed{y(t) = k e^{-\int a(t) dt}, \quad k \in \mathbb{R}}
\end{align*}

\bigskip

\subsection{Equação Diferencial Ordinária de 1ª Ordem Linear Não Homogénea}
\label{subsec: edo-ordem1-linear-nao-homogenea}

Para este tipo de equação, usamos o \textbf{método do fator integrante}.
Este método introduz uma função auxiliar $\mu(t)$, a que se chama fator integrante,
com o objetivo de transformar o lado esquerdo da equação numa derivada do produto de duas funções, $\mu(t)$ e $y(t)$.

\begin{gather*}
\boxed{y' + a(t) y = g(t)} \\
\mu(t) y' + \mu(t) a(t) y = \mu(t) g(t)
\end{gather*}

Queremos escrever o lado esquerdo como

\begin{gather*}
\frac{d}{dt} \Big[ \mu y \Big] = \mu' y + \mu y'
\end{gather*}

Para isso, é necessário que

\begin{gather*}
\mu' = \mu a(t)
\end{gather*}

Sendo esta equação uma \nameref{subsec: edo-ordem1-linear-homogenea}, tal que

\begin{gather*}
\mu' + (-a(t)) \mu = 0 \\
\implies \mu(t) = e^{\int a(t) dt}
\end{gather*}

Então, calculado o fator integrante, podemos reescrever a equação original como

\begin{align*}
& \frac{d}{dt} \Big[ \mu(t) y \Big] = \mu(t) g(t) \\
& \mu(t) y= \int \mu(t) g(t) dt \\
& \mu(t) y = \int \mu(t) g(t) dt + c, \quad c \in \mathbb{R} \\
& y(t) = \frac{1}{\mu(t)} \left( \int \mu(t) g(t) dt + c \right), \quad c \in \mathbb{R} \\
& \boxed{y(t) = e^{-\int a(t) dt} \left( \int e^{\int a(t) dt} g(t) dt + c \right), \quad c \in \mathbb{R}}
\end{align*}

\bigskip

\subsection{Equação Diferencial Ordinária de 1ª Ordem Separável}

Uma equação diferencial é dita separável se pode ser escrita na forma

\begin{gather*}
\boxed{\frac{dy}{dt} = \frac{g(t)}{f(y)}}
\end{gather*}

Separando as variáveis, obtemos

\begin{gather*}
f(y) \frac{dy}{dt} = g(t)
\end{gather*}

Integrando ambos os lados em ordem a $t$, obtemos

\begin{gather*}
\int f(y) \frac{dy}{dt} dt = \int g(t) dt
\end{gather*}

Ficando com $y(t)$ definido implicitamente por

\begin{gather*}
\boxed{\int f(y) dy = \int g(t) dt}
\end{gather*}

\smallskip

\begin{itemize}
    \item Uma Equação Separável pode ser linear ou não linear.
    \item Uma \nameref{subsec: edo-ordem1-linear-homogenea} é separável.
    \item Uma \nameref{subsec: edo-ordem1-linear-nao-homogenea} não é separável.
\end{itemize}

\bigskip

\subsection{Equação Diferencial Ordinária de 1ª Ordem Exata}
\label{subsec: edo-ordem1-exata}

\begin{gather*}
\boxed{M(t,y) + N(t,y) \frac{dy}{dt} = 0}
\end{gather*}

Esta equação diferencial é dita exata se puder ser escrita como

\begin{gather*}
\frac{d}{dt} \Big[ \phi(t,y) \Big] = 0
\end{gather*}

Usando a regra da cadeia, temos que

\begin{gather*}
\frac{d}{dt} \Big[ \phi(t,y) \Big] = \frac{\partial \phi}{\partial t} + \frac{\partial \phi}{\partial y} \frac{dy}{dt}
\end{gather*}

Logo, para que a equação seja exata, deve ser possível encontrar uma função $\phi(t,y)$ tal que

\begin{gather*}
\frac{\partial \phi}{\partial t} = M(t,y) \quad \text{ e } \quad \frac{\partial \phi}{\partial y} = N(t,y)
\end{gather*}

A solução da equação exata é dada implicitamente por

\begin{gather*}
\boxed{\phi(t,y) = c, \quad c \in \mathbb{R}}
\end{gather*}

\begin{itemize}
    \item Nota: uma equação exata pode ser linear ou não linear.
\end{itemize}

\smallskip

\paragraph{Teorema}
\begin{gather*}
\text{Existe } \phi(t,y) \text{ tal que } \frac{\partial \phi}{\partial t} = M(t,y) \text{ e } \frac{\partial \phi}{\partial y} = N(t,y) \text{ se, e só se, } \frac{\partial M}{\partial y} = \frac{\partial N}{\partial t}
\end{gather*}


\paragraph{Método de resolução de uma equação exata}
\label{par: metodo-resolucao-edo-exata}
\begin{gather*}
M(t,y) + N(t,y) \frac{dy}{dt} = 0
\end{gather*}

\begin{enumerate}
    \item Verificar se é exata.
    \begin{gather*}
    \frac{\partial M}{\partial y} = \frac{\partial N}{\partial t}
    \end{gather*}
    
    \item Se for exata, encontrar a função $\phi(t,y)$, tal que
    \begin{gather*}
    \begin{dcases}
    \frac{\partial \phi}{\partial t} = M \implies \phi = \int M dt + c_1(y) \\
    \frac{\partial \phi}{\partial y} = N \implies \phi = \int N dy + c_2(t)
    \end{dcases}
    \end{gather*}
    
    Basta calcular uma das equações, e depois determinar o seu termo $c$.
    
    Por exemplo, podemos calcular a primeira equação,
    \begin{gather*}
    \phi = \int M dt + c_1(y)
    \end{gather*}

    E depois para determinar $c_1(y)$, usamos a segunda equação com o $\phi$ encontrado.
    \begin{gather*}
    \frac{\partial \phi}{\partial y} = N \\
    \frac{\partial }{\partial y} \left( \int M dt + c_1(y) \right) = N \\
    c_1'(y) = N - \frac{\partial }{\partial y} \left( \int M dt \right)
    \end{gather*}

    \item A solução é dada implicitamente por
    \begin{gather*}
    \phi(t,y) = c, \quad c \in \mathbb{R}
    \end{gather*}
\end{enumerate}

\bigskip

\subsection{Equação Diferencial Ordinária de 1ª Ordem Redutível a Exata}

Por vezes, uma equação diferencial que não é exata pode ser transformada numa equação exata 
usando o \textbf{método do fator integrante} (visto anteriormente na resolução de uma \nameref{subsec: edo-ordem1-linear-nao-homogenea})

\begin{gather*}
\boxed{M(t,y) + N(t,y) \frac{dy}{dt} = 0}
\end{gather*}

Se a equação diferencial não for exata, ou seja, se

\begin{gather*}
\frac{\partial M}{\partial y} \neq \frac{\partial N}{\partial t}
\end{gather*}

Então vamos procurar um fator integrante $\mu(t,y)$ para tornar a equação exata.

\begin{gather*}
\mu(t,y) M(t,y) + \mu(t,y) N(t,y) \frac{dy}{dt} = 0
\end{gather*}

Para que esta nova equação seja exata, é necessário que

\begin{gather*}
\frac{\partial }{\partial y} \Big[ \mu M \Big] = \frac{\partial }{\partial t} \Big[ \mu N \Big] \\
\frac{\partial \mu}{\partial y} M + \mu \frac{\partial M}{\partial y} = \frac{\partial \mu}{\partial t} N + \mu \frac{\partial N}{\partial t}
\end{gather*}

Esta é uma equação difícil de resolver para determinar o fator integrante $\mu(t,y)$.
No entanto, em alguns casos, podemos encontrar um fator integrante que depende apenas de $t$ ou apenas de $y$.

\bigskip

Para $\mu = \mu(t)$, a equação torna-se

\begin{gather*}
0 + \mu \frac{\partial M}{\partial y} = \mu' N + \mu \frac{\partial N}{\partial t} \\
N \mu' = \mu \left( \frac{\partial M}{\partial y} - \frac{\partial N}{\partial t} \right) \\
\mu' = \mu \frac{\frac{\partial M}{\partial y} - \frac{\partial N}{\partial t}}{N} \\
\end{gather*}

Esta é uma \nameref{subsec: edo-ordem1-linear-homogenea}, que pode ser resolvida se, e só se

\begin{gather*}
\frac{\frac{\partial M}{\partial y} - \frac{\partial N}{\partial t}}{N} \quad \text{depender apenas de } t
\end{gather*}

Se assim for, ficamos então com

\begin{gather*}
\mu' + \left( - \frac{\frac{\partial M}{\partial y} - \frac{\partial N}{\partial t}}{N} \right) \mu = 0 \\
\implies \mu(t) = e^{\int \frac{\frac{\partial M}{\partial y} - \frac{\partial N}{\partial t}}{N} dt}
\end{gather*}

Similarmente, para $\mu = \mu(y)$, a equação torna-se
\begin{gather*}
\mu' M + \mu \frac{\partial M}{\partial y} = 0 + \mu \frac{\partial N}{\partial t} \\
M \mu' = \mu \left( \frac{\partial N}{\partial t} - \frac{\partial M}{\partial y} \right) \\
\mu' = \mu \frac{\frac{\partial N}{\partial t} - \frac{\partial M}{\partial y}}{M} \\
\end{gather*}

Esta é uma \nameref{subsec: edo-ordem1-linear-homogenea}, que pode ser resolvida se, e só se

\begin{gather*}
\frac{\frac{\partial N}{\partial t} - \frac{\partial M}{\partial y}}{M} \quad \text{depender apenas de } y
\end{gather*}

Se assim for, ficamos então com

\begin{gather*}
\mu' + \left( - \frac{\frac{\partial N}{\partial t} - \frac{\partial M}{\partial y}}{M} \right) \mu = 0 \\
\implies \mu(y) = e^{\int \frac{\frac{\partial N}{\partial t} - \frac{\partial M}{\partial y}}{M} dy}
\end{gather*}

\bigskip

\begin{itemize}
    \item Se conseguirmos encontrar um fator integrante $\mu(t)$ ou $\mu(y)$, fica-se então com uma \nameref{subsec: edo-ordem1-exata} que pode ser resolvida como visto anteriormente.
\end{itemize}

\paragraph{Método de resolução de uma equação redutível a exata}
\begin{gather*}
M(t,y) + N(t,y) \frac{dy}{dt} = 0
\end{gather*}

\begin{enumerate}
    \item Verificar se é exata.
    \begin{gather*}
    \frac{\partial M}{\partial y} = \frac{\partial N}{\partial t}
    \end{gather*}
    
    \begin{itemize}
        \item Se for exata, seguir o \nameref{par: metodo-resolucao-edo-exata}.
    \end{itemize}
    
    \item Se não for exata, aplicar o método do fator integrante.
    \begin{gather*}
    \mu(t,y) M(t,y) + \mu(t,y) N(t,y) \frac{dy}{dt} = 0
    \end{gather*}

    \item Procurar um fator integrante $\mu(t)$ ou $\mu(y)$.

    \begin{gather*}
    \frac{\frac{\partial M}{\partial y} - \frac{\partial N}{\partial t}}{N} \text{ depende apenas de } t \text{ então existe } \mu(t) \text{ tal que} \\
    \mu'(t) = \mu(t) \frac{\frac{\partial M}{\partial y} - \frac{\partial N}{\partial t}}{N} \\
    \mu'(t) + \left( - \frac{\frac{\partial M}{\partial y} - \frac{\partial N}{\partial t}}{N} \right) \mu(t) = 0 \\
    \implies \mu(t) = e^{\int \frac{\frac{\partial M}{\partial y} - \frac{\partial N}{\partial t}}{N} dt}
    \end{gather*}

    \begin{gather*}
    \frac{\frac{\partial N}{\partial t} - \frac{\partial M}{\partial y}}{M} \text{ depende apenas de } y \text{ então existe } \mu(y) \text{ tal que} \\
    \mu'(y) = \mu(y) \frac{\frac{\partial N}{\partial t} - \frac{\partial M}{\partial y}}{M} \\
    \mu'(y) + \left( - \frac{\frac{\partial N}{\partial t} - \frac{\partial M}{\partial y}}{M} \right) \mu(y) = 0 \\
    \implies \mu(y) = e^{\int \frac{\frac{\partial N}{\partial t} - \frac{\partial M}{\partial y}}{M} dy}
    \end{gather*}

    \medskip

    \item Se existir um fator integrante $\mu$, encontrar a função $\phi(t,y)$, tal que
    \begin{gather*}
    \begin{dcases}
    \frac{\partial \phi}{\partial t} = \mu M \implies \phi = \int \mu M dt + c_1(y) \\
    \frac{\partial \phi}{\partial y} = \mu N \implies \phi = \int \mu N dy + c_2(t)
    \end{dcases}
    \end{gather*}

    Basta calcular uma das equações, e depois determinar o seu termo $c$.

    \item A solução é dada implicitamente por:
    \begin{gather*}
    \phi(t,y) = c, \quad c \in \mathbb{R}
    \end{gather*}

\end{enumerate}

\subsection{Análise Qualitativa de Equações Diferenciais Ordinárias de 1ª Ordem}

\subsubsection{Método Iterativo de Picard}
\label{subsubsec: metodo-iterativo-picard}

\noindent
\begin{minipage}{0.3\textwidth}
\begin{gather*}
\begin{dcases}
\frac{dy}{dt} = f(t, y) \\
y(t_0) = y_0
\end{dcases}
\end{gather*}
\end{minipage}
\hfill
\begin{minipage}{0.65\textwidth}
Para o problema de valor inicial, dependendo da forma de $f(t,y)$, nem sempre
conseguimos resolver explicitamente a equação e obter uma solução.
Vamos então procurar condições que garantam a \textbf{existência} e a \textbf{unicidade}
de solução num intervalo que contenha $t_0$.
\end{minipage}

\bigskip

A ideia é aproximar sucessivamente a solução da equação por funções que não são
soluções, mas tais que o erro de aproximação é cada vez menor.

\smallskip

Se $f$ for contínua, integrando a equação entre $t_0$ e $t$, obtemos

\begin{gather*}
\begin{dcases}
\frac{dy}{dt} = f(t, y) \\
y(t_0) = y_0
\end{dcases}
\Leftrightarrow
\begin{dcases}
\int_{t_0}^{t} y'(s) \, ds = \int_{t_0}^{t} f(s, y(s)) \, ds \\
y(t_0) = y_0
\end{dcases}
\Leftrightarrow
\begin{dcases}
y(t) - y(t_0) = \int_{t_0}^{t} f(s, y(s)) \, ds \\
y(t_0) = y_0
\end{dcases}
\end{gather*}

Ou seja, o PVI é equivalente à equação integral

\begin{gather*}
\boxed{y(t) = y_0 + \int_{t_0}^{t} f(s, y(s)) \, ds}
\end{gather*}

\begin{itemize}
    \item Nota: a condição inicial já está incluída nesta equação, pois para $t = t_0$ o integral é nulo e $y(t_0) = y_0$.
\end{itemize}

\smallskip

O \textbf{método iterativo de Picard} consiste em construir uma sucessão de funções $\{y_n\}$
e ver em que casos podemos garantir que o seu limite é uma solução do PVI.

\smallskip

A equação integral anterior permite-nos definir iterativamente a sucessão $\{y_n\}$:

\newpage

\begin{itemize}
    \item $y_0(t) = y_0$ (função constante igual à condição inicial), para $n = 0$.
    \item $y_n(t) = y_0 + \int_{t_0}^{t} f(s, y_{n-1}(s)) \, ds$, para $n \geq 1$.
\end{itemize}

Se a sucessão de funções $y_n(t)$ convergir (uniformemente) num intervalo que contém $t_0$,
então o seu limite é solução do PVI nesse intervalo:

\begin{gather*}
\boxed{y(t) = \lim_{n \to \infty} y_n(t)}
\end{gather*}

\begin{itemize}
    \item Nota: esta convergência não está garantida para qualquer $f$.
    As condições sobre $f$ que a garantem são dadas pelo
    \nameref{subsubsec: teorema-existencia-unicidade-picard}.
\end{itemize}

\paragraph{Exemplo}

\begin{gather*}
\begin{dcases}
y' = y \\
y(0) = 1
\end{dcases}
%\quad \text{onde} \quad f(t,y) = y, \quad t_0 = 0, \quad y_0 = 1
\end{gather*}

Para $n = 0$:

\begin{gather*}
y_0(t) = 1
\end{gather*}

Para $n = 1$:

\begin{gather*}
y_1(t) = 1 + \int_{0}^{t} f(s, y_0(s)) \, ds = 1 + \int_{0}^{t} 1 \, ds = 1 + \Big[ s \Big]_{0}^{t} = 1 + t
\end{gather*}

Para $n = 2$:

\begin{gather*}
y_2(t) = 1 + \int_{0}^{t} f(s, y_1(s)) \, ds = 1 + \int_{0}^{t} (1 + s) \, ds
= 1 + \Big[ s + \frac{s^2}{2} \Big]_{0}^{t} = 1 + t + \frac{t^2}{2}
\end{gather*}

Para $n = 3$:

\begin{gather*}
y_3(t) = 1 + \int_{0}^{t} f(s, y_2(s)) \, ds = 1 + \int_{0}^{t} \left( 1 + s + \frac{s^2}{2} \right) ds
= 1 + \Big[ s + \frac{s^2}{2} + \frac{s^3}{3!} \Big]_{0}^{t} = 1 + t + \frac{t^2}{2} + \frac{t^3}{3!}
\end{gather*}

As iteradas sugerem que, em geral,

\begin{gather*}
\boxed{y_n(t) = \sum_{k=0}^{n} \frac{t^k}{k!}}
\end{gather*}

que é o polinómio de Taylor de ordem $n$ de $e^t$ em torno de $t = 0$.
De facto, por indução,

\begin{gather*}
y_{n+1}(t) = 1 + \int_{0}^{t} \sum_{k=0}^{n} \frac{s^k}{k!} \, ds
= 1 + \sum_{k=0}^{n} \frac{t^{k+1}}{(k+1)!}
= \sum_{k=0}^{n+1} \frac{t^k}{k!}
\end{gather*}

Neste caso, a sucessão converge para todo o $t \in \mathbb{R}$ e o seu limite é a série de Taylor de $e^t$:

\begin{gather*}
\boxed{y(t) = \lim_{n \to \infty} y_n(t) = \sum_{k=0}^{+\infty} \frac{t^k}{k!} = e^t}
\end{gather*}

\begin{itemize}
    \item Verificação: $y(t) = e^t$ satisfaz $y' = e^t = y$ e $y(0) = e^0 = 1$.
\end{itemize}

\bigskip

\subsubsection{Teorema de Existência e Unicidade de Picard}
\label{subsubsec: teorema-existencia-unicidade-picard}

\begin{gather*}
\begin{dcases}
\frac{dy}{dt} = f(t, y) \\
y(t_0) = y_0
\end{dcases}
\end{gather*}

Seja $R = [\alpha, \beta] \times [\gamma, \delta]$ um retângulo fechado que contém o ponto $(t_0, y_0)$,
com $\alpha \leq t_0 < \beta$ e $\gamma < y_0 < \delta$, e sejam

\begin{gather*}
a = \beta - t_0, \quad b_1 = \delta - y_0, \quad b_2 = y_0 - \gamma, \quad
b = \min\{b_1, b_2\}, \quad M = \max_{(t,y) \in R} |f(t,y)|
\end{gather*}

\paragraph{Teorema}
Se $f$ e $\frac{\partial f}{\partial y}$ são contínuas em $R$, então o PVI tem uma \textbf{única} solução
$y = \phi(t)$ definida no intervalo $[t_0, t_0 + \theta]$, onde

\begin{gather*}
\boxed{\theta = \min\left\{a, \frac{b}{M}\right\}}
\end{gather*}

Além disso, esta solução é o limite da sucessão das iteradas do \nameref{subsubsec: metodo-iterativo-picard}.

\begin{figure}[h]
\centering
\includegraphics[width=0.9\textwidth]{picard_rectangle}
% \caption{Retângulo $R$ e solução $y(t)$ do PVI no intervalo $[t_0, t_0 + \theta]$.}
\end{figure}

\begin{itemize}
    \item $M$ existe porque $f$ é contínua no conjunto fechado e limitado $R$ (Teorema de Weierstrass).
    Se $M = 0$, toma-se $\theta = a$.

    \item Escolha de $\theta$: enquanto a solução está em $R$, temos $|y'(t)| = |f(t, y(t))| \leq M$, logo
    \begin{gather*}
    |y(t) - y_0| \leq M (t - t_0) \leq M \theta \leq b, \quad t \in [t_0, t_0 + \theta]
    \end{gather*}
    Ou seja, $\theta \leq a$ garante que $t$ não sai de $[\alpha, \beta]$ e $\theta \leq \frac{b}{M}$ garante
    que $y$ não sai de $[y_0 - b, y_0 + b] \subseteq [\gamma, \delta]$.
    Usa-se $b = \min\{b_1, b_2\}$ porque a solução pode tanto subir como descer.

    \item O teorema é \textbf{local}: só garante a solução perto de $t_0$, e $\theta$ pode ser muito menor que $a$.

    \item Se $\alpha < t_0$, o mesmo resultado vale à esquerda de $t_0$, no intervalo $[t_0 - \theta, t_0]$,
    tomando $a = t_0 - \alpha$.

    \item Se apenas $f$ for contínua, garante-se a existência de solução, mas não a unicidade.
\end{itemize}

\section{Equações Diferenciais Ordinárias de Ordem Superior a Um}

\subsection{Equação Diferencial Ordinária de 2ª Ordem Linear}
\label{subsec: edo-ordem2-linear}

\begin{gather*}
\boxed{\frac{d^2y}{dt^2} + p(t)\frac{dy}{dt} + q(t)y = g(t)}
\end{gather*}

\begin{itemize}
    \item A equação é homogénea se $g(t) = 0$ e não homogénea se $g(t) \neq 0$.
\end{itemize}

Para determinar uma única solução precisamos de 2 condições iniciais, por exemplo,
$y(t_0) = y_0$ e $y'(t_0) = y_0'$.

\paragraph{Teorema de Existência e Unicidade}

\begin{gather*}
\begin{dcases}
y'' + p(t)y' + q(t)y = g(t) \\
y(t_0) = y_0 \\
y'(t_0) = y_0'
\end{dcases}
\end{gather*}

Se $p(t)$, $q(t)$ e $g(t)$ são contínuas num intervalo aberto $I$ que contém $t_0$, então o PVI tem
uma \textbf{única} solução, definida em todo o intervalo $I$.

\begin{itemize}
    \item Ao contrário do \nameref{subsubsec: teorema-existencia-unicidade-picard}, este resultado não é
    local: a solução existe em todo o intervalo $I$ onde os coeficientes são contínuos.
\end{itemize}

\paragraph{Wronskiano e Independência Linear}

Dadas $n$ funções $y_1(t), y_2(t), \ldots, y_n(t)$, $(n-1)$ vezes diferenciáveis num intervalo $I$,
o \textbf{Wronskiano} destas funções é o determinante da matriz composta por essas funções e
pelas suas derivadas:

\begin{gather*}
W(y_1, y_2, \ldots, y_n)(t) =
\begin{vmatrix}
y_1 & y_2 & \cdots & y_n \\
y_1' & y_2' & \cdots & y_n' \\
\vdots & \vdots & \ddots & \vdots \\
y_1^{(n-1)} & y_2^{(n-1)} & \cdots & y_n^{(n-1)}
\end{vmatrix}
\end{gather*}

\begin{gather*}
\boxed{\exists \, t_0 \in I : W(y_1, y_2, \ldots, y_n)(t_0) \neq 0 \implies y_1, y_2, \ldots, y_n
\text{ são linearmente independentes em } I}
\end{gather*}

\paragraph{Teorema}

Sejam $y_1$ e $y_2$ duas soluções da equação homogénea $y'' + p(t)y' + q(t)y = 0$, com $p(t)$ e $q(t)$
contínuas num intervalo aberto $I$. Se

\begin{gather*}
W(y_1, y_2)(t) =
\begin{vmatrix}
y_1 & y_2 \\
y_1' & y_2'
\end{vmatrix}
= y_1 y_2' - y_2 y_1' \neq 0 \text{ em } I
\end{gather*}

então

\begin{gather*}
\boxed{y(t) = c_1 y_1(t) + c_2 y_2(t), \quad c_1, c_2 \in \mathbb{R}}
\end{gather*}

é a \textbf{solução geral} da equação.

\begin{itemize}
    \item $\{y_1, y_2\}$ é chamado um \textbf{conjunto fundamental de soluções}.
    \item Qualquer combinação linear de soluções da equação homogénea é também solução
    (princípio da sobreposição).
    \item Para duas soluções da equação homogénea, o Wronskiano ou é nulo em todo o $I$, ou nunca se anula
    em $I$. Por isso, basta verificar que $W(y_1, y_2)(t_0) \neq 0$ num único ponto $t_0 \in I$.
\end{itemize}

\paragraph{Estratégia Geral para resolver uma equação não homogénea}

\begin{gather*}
\boxed{y'' + p(t)y' + q(t)y = g(t)}
\end{gather*}

\begin{enumerate}
    \item Resolver a equação homogénea associada, $y'' + p(t)y' + q(t)y = 0$, obtendo
    \begin{gather*}
    y_h = c_1 y_1 + c_2 y_2, \quad c_1, c_2 \in \mathbb{R}
    \end{gather*}

    \item Procurar uma solução particular $y_p$ da equação original (não homogénea).
    Para obter $y_p$ podemos usar 2 métodos:
    \begin{itemize}
        \item \textbf{\nameref{subsec: metodo-coeficientes-indeterminados}} (ou Método do Aniquilador)
        \begin{itemize}
            \item Usa aniquiladores.
            \item Fácil de aplicar (só envolve derivadas).
            \item Várias restrições: serve apenas para equações com coeficientes constantes e quando $g(t)$
            é uma soma de produtos de polinómios, exponenciais, senos e cossenos.
        \end{itemize}
        \item \textbf{Método de Variação dos Parâmetros}
        \begin{itemize}
            \item Mais difícil de aplicar (envolve primitivas).
            \item Poucas restrições: basta conhecer $y_1$, $y_2$ e que $g(t)$ seja contínua.
            \item Não é apresentado nestes resumos.
        \end{itemize}
    \end{itemize}

    \item A solução geral é dada por
    \begin{gather*}
    \boxed{y = y_h + y_p = c_1 y_1 + c_2 y_2 + y_p, \quad c_1, c_2 \in \mathbb{R}}
    \end{gather*}
\end{enumerate}

\bigskip

\subsection{Equação Diferencial Ordinária de 2ª Ordem Linear Homogénea com Coeficientes Constantes}
\label{subsec: edo-ordem2-linear-homogenea-coef-constantes}

\begin{gather*}
\boxed{ay'' + by' + cy = 0, \quad a, b, c \in \mathbb{R}, \quad a \neq 0}
\end{gather*}

Como a derivada de uma exponencial é proporcional à própria exponencial, faz sentido
procurar soluções da forma $y = e^{rt}$, onde $r$ é um parâmetro a determinar:

\begin{align*}
& ay'' + by' + cy = 0 \\
\Leftrightarrow \ & a \left( e^{rt} \right)'' + b \left( e^{rt} \right)' + c e^{rt} = 0 \\
\Leftrightarrow \ & a r^2 e^{rt} + b r e^{rt} + c e^{rt} = 0 \\
\Leftrightarrow \ & \left( a r^2 + b r + c \right) e^{rt} = 0 \\
\Leftrightarrow \ & a r^2 + b r + c = 0 \quad \text{pois } e^{rt} \neq 0 \\
\Leftrightarrow \ & r = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a} = \frac{-b \pm \sqrt{\Delta}}{2a}
\end{align*}

\begin{itemize}
    \item $a r^2 + b r + c = 0$ é a \textbf{equação característica} e $a r^2 + b r + c$ o
    \textbf{polinómio característico}.
\end{itemize}

\paragraph{$\Delta > 0$: duas raízes reais distintas}

\begin{gather*}
r_1 = \frac{-b - \sqrt{\Delta}}{2a} \quad \text{e} \quad r_2 = \frac{-b + \sqrt{\Delta}}{2a}
\end{gather*}

As duas raízes correspondem a duas soluções da equação:

\begin{gather*}
\begin{dcases}
y_1 = e^{r_1 t} \\
y_2 = e^{r_2 t}
\end{dcases}
\end{gather*}

Tendo que

\begin{gather*}
W(y_1, y_2)(t) =
\begin{vmatrix}
e^{r_1 t} & e^{r_2 t} \\
r_1 e^{r_1 t} & r_2 e^{r_2 t}
\end{vmatrix}
= (r_2 - r_1) e^{(r_1 + r_2)t} \neq 0 \quad \text{pois } r_1 \neq r_2
\end{gather*}

logo as duas soluções são linearmente independentes e a solução geral da equação é

\begin{gather*}
\boxed{y(t) = c_1 e^{r_1 t} + c_2 e^{r_2 t}, \quad c_1, c_2 \in \mathbb{R}}
\end{gather*}

\paragraph{$\Delta = 0$: uma raiz real dupla}

\begin{gather*}
r = -\frac{b}{2a}
\end{gather*}

Esta raiz corresponde a uma solução da equação, $y_1 = e^{rt}$.
Precisamos de mais uma solução, linearmente independente de $y_1$, para aplicar o Teorema anterior
e obter a solução geral.
Para isso usa-se o \textbf{método de redução de ordem} (que não é apresentado aqui), obtendo-se

\begin{gather*}
y_2 = t e^{rt}
\end{gather*}

Tendo que

\begin{gather*}
W(y_1, y_2)(t) =
\begin{vmatrix}
e^{rt} & t e^{rt} \\
r e^{rt} & e^{rt} + r t e^{rt}
\end{vmatrix}
= e^{2rt} \neq 0
\end{gather*}

logo as duas soluções são linearmente independentes e a solução geral da equação é

\begin{gather*}
\boxed{y(t) = c_1 e^{rt} + c_2 t e^{rt}, \quad c_1, c_2 \in \mathbb{R}}
\end{gather*}

\paragraph{$\Delta < 0$: duas raízes complexas conjugadas}

\begin{gather*}
r_1 = \alpha - i \beta \quad \text{e} \quad r_2 = \alpha + i \beta,
\quad \text{com } \alpha = -\frac{b}{2a} \text{ e } \beta = \frac{\sqrt{-\Delta}}{2a}
\end{gather*}

As duas raízes correspondem a duas soluções da equação:

\begin{gather*}
\begin{dcases}
y_1 = e^{(\alpha - i \beta) t} \\
y_2 = e^{(\alpha + i \beta) t}
\end{dcases}
\end{gather*}

No entanto, estas soluções são complexas. Como a equação tem coeficientes reais, queremos soluções reais.
Usando a fórmula de Euler, $e^{i \theta} = \cos(\theta) + i \sin(\theta)$, temos

\begin{gather*}
e^{(\alpha \pm i \beta) t} = e^{\alpha t} \left( \cos(\beta t) \pm i \sin(\beta t) \right)
\end{gather*}

e (sem demonstrar) as partes real e imaginária desta exponencial são também soluções da equação:

\begin{gather*}
\begin{dcases}
y_1 = \operatorname{Re}\left[ e^{(\alpha + i \beta) t} \right] = e^{\alpha t} \cos(\beta t) \\
y_2 = \operatorname{Im}\left[ e^{(\alpha + i \beta) t} \right] = e^{\alpha t} \sin(\beta t)
\end{dcases}
\end{gather*}

Tendo que

\begin{gather*}
W(y_1, y_2)(t) =
\begin{vmatrix}
e^{\alpha t} \cos(\beta t) & e^{\alpha t} \sin(\beta t) \\
e^{\alpha t} \left( \alpha \cos(\beta t) - \beta \sin(\beta t) \right) &
e^{\alpha t} \left( \alpha \sin(\beta t) + \beta \cos(\beta t) \right)
\end{vmatrix}
= \beta e^{2 \alpha t} \neq 0 \quad \text{pois } \beta \neq 0
\end{gather*}

logo as duas soluções são linearmente independentes e a solução geral da equação é

\begin{gather*}
\boxed{y(t) = c_1 e^{\alpha t} \cos(\beta t) + c_2 e^{\alpha t} \sin(\beta t), \quad c_1, c_2 \in \mathbb{R}}
\end{gather*}

\bigskip

\subsection{Equação Diferencial Ordinária de Ordem Superior a Dois Linear Homogénea com Coeficientes Constantes}
\label{subsec: edo-ordem-n-linear-homogenea-coef-constantes}

A forma geral de uma equação diferencial ordinária linear homogénea de ordem $n$ é

\begin{gather*}
a_n(t) \frac{d^n y}{dt^n} + a_{n-1}(t) \frac{d^{n-1} y}{dt^{n-1}} + \ldots + a_1(t) \frac{dy}{dt} + a_0(t) y = 0
\end{gather*}

No entanto, vamos focar-nos em equações com coeficientes constantes:

\begin{gather*}
\boxed{a_n y^{(n)} + a_{n-1} y^{(n-1)} + \ldots + a_1 y' + a_0 y = 0, \quad a_0, \ldots, a_n \in \mathbb{R}, \quad a_n \neq 0}
\end{gather*}

Tal como na \nameref{subsec: edo-ordem2-linear-homogenea-coef-constantes}, procuramos soluções da
forma $y = e^{rt}$, obtendo a equação característica

\begin{gather*}
a_n r^n + a_{n-1} r^{n-1} + \ldots + a_1 r + a_0 = 0
\end{gather*}

Para obter a solução geral de uma equação de ordem $n$, precisamos de $n$ soluções linearmente
independentes, $y_1, \ldots, y_n$ (ou seja, $W(y_1, \ldots, y_n) \neq 0$), e então

\begin{gather*}
\boxed{y(t) = c_1 y_1(t) + c_2 y_2(t) + \ldots + c_n y_n(t), \quad c_1, \ldots, c_n \in \mathbb{R}}
\end{gather*}

A equação característica tem $n$ raízes (contando com a multiplicidade), que dão origem às $n$ soluções
linearmente independentes.

\smallskip

Uma raiz real $r$ de multiplicidade $m$ dá origem a $m$ soluções:

\begin{itemize}
    \item $y_1 = e^{rt}$
    \item $y_2 = t e^{rt}$
    \item \ldots
    \item $y_m = t^{m-1} e^{rt}$
\end{itemize}

Um par de raízes complexas conjugadas $\alpha \pm i \beta$, cada uma de multiplicidade $m$, dá origem a
$2m$ soluções:

\begin{itemize}
    \item $y_1 = e^{\alpha t} \cos(\beta t), \quad y_2 = t e^{\alpha t} \cos(\beta t), \quad \ldots, \quad
    y_m = t^{m-1} e^{\alpha t} \cos(\beta t)$
    \item $y_{m+1} = e^{\alpha t} \sin(\beta t), \quad y_{m+2} = t e^{\alpha t} \sin(\beta t), \quad \ldots, \quad
    y_{2m} = t^{m-1} e^{\alpha t} \sin(\beta t)$
\end{itemize}

\begin{itemize}
    \item Nota: para encontrar as raízes, fatoriza-se o polinómio característico com a Regra de Ruffini.
    Se os coeficientes forem inteiros, as raízes inteiras são divisores de $a_0$.
\end{itemize}

\paragraph{Exemplo}

\begin{gather*}
y''' + 3y'' + 3y' + y = 0
\end{gather*}

A equação característica é

\begin{gather*}
r^3 + 3r^2 + 3r + 1 = 0
\end{gather*}

Testando os divisores de $a_0 = 1$, vemos que $r = -1$ é raiz. Aplicando a Regra de Ruffini duas vezes:

\begin{gather*}
\begin{array}{r|rrrr}
   & 1 & 3  & 3  & 1  \\
-1 &   & -1 & -2 & -1 \\
\hline
   & 1 & 2  & 1  & 0
\end{array}
\qquad \qquad
\begin{array}{r|rrr}
   & 1 & 2  & 1  \\
-1 &   & -1 & -1 \\
\hline
   & 1 & 1  & 0
\end{array}
\end{gather*}

\begin{align*}
& r^3 + 3r^2 + 3r + 1 = 0 \\
\Leftrightarrow \ & (r + 1)(r^2 + 2r + 1) = 0 \\
\Leftrightarrow \ & (r + 1)(r + 1)(r + 1) = 0 \\
\Leftrightarrow \ & (r + 1)^3 = 0
\end{align*}

Logo, $r = -1$ é uma raiz real de multiplicidade 3, e a solução geral é

\begin{gather*}
\boxed{y(t) = c_1 e^{-t} + c_2 t e^{-t} + c_3 t^2 e^{-t}, \quad c_1, c_2, c_3 \in \mathbb{R}}
\end{gather*}

\subsection{Método dos Coeficientes Indeterminados}
\label{subsec: metodo-coeficientes-indeterminados}

Seja $D$ o operador derivada:

\begin{itemize}
    \item $Dy = y'$
    \item $D^2 y = y''$
    \item \ldots
    \item $D^n y = y^{(n)}$
\end{itemize}

Um \textbf{aniquilador} de uma função $f(t)$ é um operador, dado por um polinómio em $D$ com coeficientes
constantes, que, quando aplicado a $f(t)$, produz a função nula.

\smallskip

Este método determina o aniquilador de $g(t)$ para transformar a equação original, não homogénea de
ordem 2, numa equação homogénea de ordem superior a dois (que se resolve como na
\nameref{subsec: edo-ordem-n-linear-homogenea-coef-constantes}).

\paragraph{Aniquiladores de funções elementares}

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|}
\hline
Função $f(t)$ & Aniquilador \\
\hline
$t^m$ & $D^{m+1}$ \\
$e^{\alpha t}$ & $D - \alpha$ \\
$t e^{\alpha t}$ & $(D - \alpha)^2$ \\
$t^m e^{\alpha t}$ & $(D - \alpha)^{m+1}$ \\
$\cos(\beta t)$ & $D^2 + \beta^2$ \\
$\sin(\beta t)$ & $D^2 + \beta^2$ \\
$t^m \cos(\beta t)$ & $(D^2 + \beta^2)^{m+1}$ \\
$t^m \sin(\beta t)$ & $(D^2 + \beta^2)^{m+1}$ \\
$e^{\alpha t} \cos(\beta t)$ & $(D - \alpha)^2 + \beta^2$ \\
$e^{\alpha t} \sin(\beta t)$ & $(D - \alpha)^2 + \beta^2$ \\
$t^m e^{\alpha t} \cos(\beta t)$ & $\left[ (D - \alpha)^2 + \beta^2 \right]^{m+1}$ \\
$t^m e^{\alpha t} \sin(\beta t)$ & $\left[ (D - \alpha)^2 + \beta^2 \right]^{m+1}$ \\
\hline
\end{tabular}
\end{center}

\begin{itemize}
    \item O aniquilador de $t^m e^{\alpha t}$ também aniquila $e^{\alpha t}, t e^{\alpha t}, \ldots,
    t^{m-1} e^{\alpha t}$ (e analogamente para senos e cossenos).
    Por exemplo, um polinómio de grau $m$ é aniquilado por $D^{m+1}$.
\end{itemize}

O aniquilador de uma soma de funções é o produto dos aniquiladores de cada uma:

\begin{gather*}
f(t) = u(t) + v(t) \implies D_f = D_u D_v
\end{gather*}

\begin{itemize}
    \item Se $D_u$ e $D_v$ tiverem fatores em comum, basta incluí-los uma vez. Por exemplo,
    $\cos(\beta t) + \sin(\beta t)$ é aniquilada por $D^2 + \beta^2$.
\end{itemize}

\paragraph{Aplicar o Método}

\begin{gather*}
\boxed{ay'' + by' + cy = g(t)}
\end{gather*}

\begin{enumerate}
    \item Determinar $y_h = c_1 y_1 + c_2 y_2$, resolvendo $ay'' + by' + cy = 0$
    (com as raízes $r_1$ e $r_2$ de $ar^2 + br + c$).

    \item Determinar o aniquilador de $g(t)$, $D_g$.

    \item Escrever a equação com o operador $D$ e aplicar o aniquilador:
    \begin{gather*}
    aD^2 y + bDy + cy = g(t)
    \Leftrightarrow (aD^2 + bD + c) y = g(t)
    \implies D_g (aD^2 + bD + c) y = 0
    \end{gather*}
    Obtém-se uma equação homogénea de ordem superior a dois, cuja equação característica é
    \begin{gather*}
    P_g(r) (ar^2 + br + c) = 0
    \end{gather*}
    onde $P_g(r)$ é o polinómio que se obtém substituindo $D$ por $r$ em $D_g$.

    \item Determinar as raízes de $P_g(r)$ (as de $ar^2 + br + c$ já são conhecidas do passo 1).
    Se uma raiz de $P_g(r)$ também for raiz de $ar^2 + br + c$, as multiplicidades somam-se.

    \item Obter a forma de $y_p$. A solução geral da equação homogénea de ordem superior é
    \begin{gather*}
    y = c_1 y_1 + c_2 y_2 + A y_3 + B y_4 + \ldots
    \end{gather*}
    onde $y_3, y_4, \ldots$ são as soluções que vêm das raízes de $P_g(r)$.
    Como $c_1 y_1 + c_2 y_2 = y_h$ é solução da equação homogénea (ou seja, dá $0$ e não $g(t)$),
    excluem-se esses termos:
    \begin{gather*}
    \boxed{y_p = A y_3 + B y_4 + \ldots}
    \end{gather*}

    % Usam-se, de propósito, os coeficientes A, B, C, etc. e não c_1, c_2, etc.

    \item Determinar os coeficientes $A, B, \ldots$: calcular $y_p'$ e $y_p''$, substituir em
    $ay_p'' + by_p' + cy_p = g(t)$ e igualar os coeficientes dos termos semelhantes.

    \item A solução geral é $y = y_h + y_p$.
\end{enumerate}

\begin{itemize}
    \item Nota: por exemplo, em $y'' - y = e^t$, temos $ar^2 + br + c = (r - 1)(r + 1)$ e $D_g = D - 1$,
    logo $(r - 1)^2 (r + 1) = 0$ e as soluções são $e^t$, $t e^t$ e $e^{-t}$.
    Excluindo os termos de $y_h = c_1 e^t + c_2 e^{-t}$, fica $y_p = A t e^t$ (e não $A e^t$).
\end{itemize}
